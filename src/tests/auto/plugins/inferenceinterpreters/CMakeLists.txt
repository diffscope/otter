# The tests of single plugins, one directory per plugin. A test covers one header of the plugin and
# links the objects of the plugin, <target>_objects, as the test of a class of a library links the
# library.
#
# The test of an interpreter registers the interpreter it links as a static plugin, with a copy of
# the plugin class that main.cpp of the plugin declares and with the plugin.json of the plugin. A
# unit adds the static plugins of a category when the category is registered, before it scans any
# plugin directory, so discovery selects the class under test and no library of the plugin tree is
# loaded.

# The plugin that the analysis tests use in place of a model, one directory per plugin
add_subdirectory(stub)

# otter_add_interpreter_test(<source> <directory> [<libraries>...])
#
# Adds the test of the interpreter of inferenceinterpreters/<directory>, which links <libraries>,
# the objects of that plugin among them, and reads the plugin.json of that directory through
# OTTER_TEST_PLUGIN_MANIFEST. The test reports with OTTER_TEST_SKIP_EXIT_CODE that it did not run.
function(otter_add_interpreter_test _source _directory)
    get_filename_component(_name ${_source} NAME_WE)
    add_auto_test(${_source} ${ARGN})
    if(NOT TARGET ${_name})
        return()
    endif()
    set(_manifest "${OTTER_SOURCE_DIR}/src/plugins/inferenceinterpreters/${_directory}/plugin.json")
    target_compile_definitions(${_name} PRIVATE
        OTTER_TEST_SKIP_EXIT_CODE=${OTTER_TEST_SKIP_EXIT_CODE}
        OTTER_TEST_PLUGIN_MANIFEST="${_manifest}"
    )
    set_tests_properties(${_name} PROPERTIES SKIP_RETURN_CODE ${OTTER_TEST_SKIP_EXIT_CODE})
endfunction()

# The model fixtures are real ONNX graphs generated by a script, so they reside in the build tree
# and are generated by the build. Generation requires a Python interpreter with the onnx and numpy
# modules. Without such an interpreter, the model tests that need the fixtures are registered as
# disabled, which ctest reports, rather than detecting the missing fixtures at run time and
# passing.
find_package(Python3 COMPONENTS Interpreter QUIET)
set(_otter_can_make_fixtures OFF)
if(Python3_Interpreter_FOUND)
    execute_process(
        COMMAND "${Python3_EXECUTABLE}" -c "import onnx, numpy"
        RESULT_VARIABLE _otter_onnx_missing
        OUTPUT_QUIET ERROR_QUIET
    )
    if(_otter_onnx_missing EQUAL 0)
        set(_otter_can_make_fixtures ON)
    endif()
endif()

set(OTTER_TEST_FIXTURE_DIR "${CMAKE_BINARY_DIR}/fixtures")
if(OTTER_HAS_DSINFER)
    if(_otter_can_make_fixtures)
        add_custom_command(
            OUTPUT "${OTTER_TEST_FIXTURE_DIR}/.stamp"
            COMMAND "${Python3_EXECUTABLE}" "${OTTER_SOURCE_DIR}/scripts/make-model-fixtures.py"
                    --output "${OTTER_TEST_FIXTURE_DIR}"
            COMMAND "${CMAKE_COMMAND}" -E touch "${OTTER_TEST_FIXTURE_DIR}/.stamp"
            DEPENDS "${OTTER_SOURCE_DIR}/scripts/make-model-fixtures.py"
            COMMENT "Generating the model fixtures"
            VERBATIM
        )
        add_custom_target(otter_model_fixtures DEPENDS "${OTTER_TEST_FIXTURE_DIR}/.stamp")
        message(STATUS "Model fixtures: generated into ${OTTER_TEST_FIXTURE_DIR}")
    else()
        message(STATUS "Model fixtures: no Python with onnx and numpy; test_RmvpeInterpreter, "
                       "test_GameInterpreter, test_HfaInterpreter and test_TifaInterpreter are "
                       "disabled")
    endif()
endif()

# The drivers directory of the installed dsinfer, the parent of the bundle of the imported driver
# target. Without the driver it is empty, and the model tests report a skip.
set(_otter_driver_plugin_dir "")
if(TARGET synthrt::onnxdriver)
    set(_otter_driver_plugin_dir "$<TARGET_FILE_DIR:synthrt::onnxdriver>/..")
endif()

# Missing fixtures, driver or runtime at run time are reported with the skip status, which ctest
# shows as a skip rather than a pass.
foreach(_model_test IN ITEMS test_RmvpeInterpreter test_GameInterpreter test_HfaInterpreter
                             test_TifaInterpreter)
    if(TARGET ${_model_test})
        if(_otter_can_make_fixtures)
            add_dependencies(${_model_test} otter_model_fixtures)
        else()
            set_tests_properties(${_model_test} PROPERTIES DISABLED TRUE)
        endif()
        target_compile_definitions(${_model_test} PRIVATE
            OTTER_TEST_FIXTURE_DIR="${OTTER_TEST_FIXTURE_DIR}"
            OTTER_TEST_ONNXRUNTIME_DIR="${OTTER_ONNXRUNTIME_DIR}"
            OTTER_TEST_DRIVER_PLUGIN_DIR="${_otter_driver_plugin_dir}"
        )
    endif()
endforeach()

# The ggml tests load packages they write themselves and need no fixtures, packages directory or
# driver. Their execution cases run against a stand-in command line tool that answers the engines'
# protocols with canned output, which the build compiles beside the tests; the real-engine cases
# name the engine through the environment and report themselves as a skip without it.
add_executable(otterfakecli fakecli.cpp)
target_compile_features(otterfakecli PRIVATE cxx_std_17)
foreach(_ggml_test IN ITEMS test_GameGgmlInterpreter test_TifaGgmlInterpreter test_GameGgmlInterpreter_RealEngine test_TifaGgmlInterpreter_RealEngine)
    if(TARGET ${_ggml_test})
        add_dependencies(${_ggml_test} otterfakecli)
        target_compile_definitions(${_ggml_test} PRIVATE
            OTTER_TEST_FAKE_CLI="$<TARGET_FILE:otterfakecli>"
        )
    endif()
endforeach()
